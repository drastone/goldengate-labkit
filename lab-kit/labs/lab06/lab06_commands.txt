# Target DB: <BASE_SCN>을 위에서 확인한 실제 SCN으로 바꾼다.
docker exec -it demo_otrg_23ai bash
sqlplus soe/soe@orclpdb1
create database link to_src connect to soe identified by soe using '(DESCRIPTION=(ADDRESS=(PROTOCOL=TCP)(HOST=oracle_src_23ai)(PORT=1521))(CONNECT_DATA=(SERVICE_NAME=ORCLPDB1)))';
select 1 as ping from dual@to_src;
truncate table soe.employee;
insert into soe.employee select * from soe.employee@to_src as of scn <BASE_SCN>;
commit;
exit
exit

# OGG Hub: D02 parameter에는 다음 네 줄을 저장한다.
docker exec -it -u ogg demo_ogg_23ai adminclient
connect http://localhost as oggadmin !
edit params d02
REPLICAT D02
USERIDALIAS trg_ogg DOMAIN OracleGoldenGate
DISCARDFILE D02.dsc, APPEND
MAP SOE.employee, TARGET SOE.employee;

view params d02
dblogin useridalias trg_ogg domain OracleGoldenGate
add replicat d02 exttrail ca
start replicat d02 atcsn <BASE_SCN>
send d02 status
stats d02

# D01과 D02가 모두 At EOF이고 Sequence/RBA가 같을 때만 전환한다.
stop c01
send d01 status
send d02 status
stop d02
delete replicat d02
stop d01
edit params d01

# D01 parameter에서 다음 줄을 삭제한 뒤 저장한다.
TABLEEXCLUDE soe.employee;

start d01
start c01
info all
exit
