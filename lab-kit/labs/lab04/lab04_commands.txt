archive log list;

alter session set container=ORCLPDB1;
alter profile default limit password_life_time unlimited;

create tablespace SOE
  datafile size 1G
  autoextend on next 100M maxsize 5G;

create user soe identified by soe
  default tablespace SOE
  temporary tablespace TEMP
  profile default
  account unlock;

grant CONNECT, RESOURCE, CREATE VIEW, ALTER SESSION,
      CREATE DATABASE LINK,
      ANALYZE ANY DICTIONARY, ANALYZE ANY,
      CREATE JOB, MANAGE SCHEDULER, MANAGE ANY QUEUE to SOE;

grant EXECUTE ON dbms_lock to SOE;
grant SELECT ON v_$database to SOE;
alter user soe quota unlimited on SOE;

create tablespace OGG
  datafile size 100M
  autoextend on next 50M maxsize 1G
  extent management local uniform size 1M
  segment space management auto;

create user ogg identified by ogg
  default tablespace OGG
  temporary tablespace TEMP
  profile default
  account unlock;

alter user ogg quota unlimited on OGG;
grant CONNECT, RESOURCE to ogg;
grant OGG_CAPTURE to ogg;

alter session set container=CDB$ROOT;
show parameter enable_goldengate_replication
archive log list;

select supplemental_log_data_min,
       supplemental_log_data_pk,
       supplemental_log_data_ui,
       supplemental_log_data_fk,
       supplemental_log_data_all
from v$database;

alter session set container=ORCLPDB1;

select username, account_status, default_tablespace, profile
from dba_users
where username in ('SOE', 'OGG')
order by username;

select tablespace_name, autoextensible, bytes/1024/1024 as mb,
       maxbytes/1024/1024 as max_mb
from dba_data_files
where tablespace_name in ('SOE', 'OGG')
order by tablespace_name;
